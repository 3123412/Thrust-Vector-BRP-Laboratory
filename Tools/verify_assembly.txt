var g=BRP.LaboratoryGame.Instance;
var a=g.Assembly;
var report=new BRP.Editor.VerificationSuite.Report{utc=System.DateTime.UtcNow.ToString("O")};
void Require(bool pass,string message){if(!pass)throw new System.InvalidOperationException(message);}
void Check(string name,System.Action test){try{test();report.checks.Add(new BRP.Editor.VerificationSuite.CheckResult{name=name,passed=true});report.passed++;}catch(System.Exception e){report.checks.Add(new BRP.Editor.VerificationSuite.CheckResult{name=name,details=e.Message});report.failed++;}}
g.UI.CloseDialog();g.StartSandbox();g.ClearCraft();
BRP.PartView Place(BRP.PartKind kind,UnityEngine.Vector3 location,UnityEngine.Quaternion rotation){var v=a.Spawn(g.Catalog.Find(kind));v.transform.localPosition=location;v.transform.localRotation=rotation;Require(a.TryTarget(v,out var target),"No snap target for "+kind);a.Attach(v,target);return v;}
BRP.PartView root=null,engine=null;
Check("catalog parts snap to structural ports",()=>{root=Place(BRP.PartKind.FirstStage,UnityEngine.Vector3.zero,UnityEngine.Quaternion.identity);engine=Place(BRP.PartKind.BoosterEngine,new UnityEngine.Vector3(0,-1.1f,0),UnityEngine.Quaternion.identity);Place(BRP.PartKind.Fairing,new UnityEngine.Vector3(0,1.8f,0),UnityEngine.Quaternion.identity);Require(a.Blueprint.Validate(g.Catalog).Count==0&&a.Blueprint.parts.Count==3,"Broken assembly");});
Check("detach and reattach preserve subtree",()=>{var old=engine.Placement.position;var group=a.Detach(engine);engine.transform.localPosition=old;Require(a.TryTarget(engine,out var target),"Detached engine cannot return");a.Attach(engine,target,group);Require(a.Blueprint.parts.Count==3&&a.detachCount==1&&a.snapCount==4,"Incorrect transaction");});
Check("paint undo and redo restore state",()=>{var original=root.Placement.color;a.Paint(root,UnityEngine.Color.red);a.Undo();Require(a.Blueprint.parts[0].color==original,"Undo paint failed");a.Redo();Require(a.Blueprint.parts[0].color==UnityEngine.Color.red,"Redo paint failed");});
Check("four-way fin symmetry has distinct ports",()=>{var fin=Place(BRP.PartKind.Fin,new UnityEngine.Vector3(.45f,.6f,0),UnityEngine.Quaternion.identity);a.Symmetry(fin);Require(a.Blueprint.parts.Count(p=>g.Catalog.Find(p.partId).kind==BRP.PartKind.Fin)==4,"Missing fin copies");Require(a.Blueprint.Validate(g.Catalog).Count==0,"Symmetry occupied duplicate ports");});
Check("full subtree detach and undo",()=>{root=a.views.First(v=>v.Attached&&v.Definition.kind==BRP.PartKind.FirstStage);int before=a.Blueprint.parts.Count;var group=a.Detach(root);Require(a.Blueprint.parts.Count==0&&group.Count==before,"Subtree split");a.Undo();Require(a.Blueprint.parts.Count==before,"Undo subtree failed");});
Check("serialized menu command opens and closes help",()=>{var command=g.UI.GetComponentsInChildren<BRP.LabButton>(true).First(b=>b.command=="help");command.GetComponent<UnityEngine.UI.Button>().onClick.Invoke();Require(g.UI.ModalOpen,"Button listener missing");g.UI.CloseDialog();Require(!g.UI.ModalOpen,"Dialog remains active");});
Check("authored UI and VR references exist",()=>{Require(g.flightCamera!=null&&g.flightMonitor!=null,"Flight monitor missing");Require(g.UI.GetComponentsInChildren<BRP.AttitudeIndicator>(true).Single().GetComponent<UnityEngine.CanvasRenderer>()!=null,"Attitude renderer missing");Require(UnityEngine.Object.FindObjectsByType<UnityEngine.XR.Interaction.Toolkit.Locomotion.Teleportation.TeleportationArea>(UnityEngine.FindObjectsInactive.Include,UnityEngine.FindObjectsSortMode.None).Any(t=>t.teleportationProvider!=null),"Teleport provider not connected");});
g.LoadPreset(2);g.FitAssembly();
System.IO.File.WriteAllText("Artifacts/assembly-verification.json",UnityEngine.JsonUtility.ToJson(report,true));
return report;
